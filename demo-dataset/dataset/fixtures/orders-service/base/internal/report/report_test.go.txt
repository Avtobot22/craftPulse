package report

import "testing"

func TestTitle(t *testing.T) {
	tests := []struct {
		name string
		give string
		want string
	}{
		{name: "slug", give: "monthly-revenue", want: "Monthly Revenue"},
		{name: "single word", give: "refunds", want: "Refunds"},
		{name: "spaces", give: " daily-orders ", want: "Daily Orders"},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			if got := Title(tt.give); got != tt.want {
				t.Fatalf("Title(%q) = %q, want %q", tt.give, got, tt.want)
			}
		})
	}
}
