package order

import (
	"errors"
	"fmt"
)

// ErrEmpty is returned for an order without lines.
var ErrEmpty = errors.New("order has no lines")

// Line is one position of an order; prices are in kopecks.
type Line struct {
	SKU       string `json:"sku"`
	Quantity  int    `json:"quantity"`
	UnitPrice int64  `json:"unit_price"`
}

// Order is a customer order with an optional percentage discount.
type Order struct {
	ID              string `json:"id"`
	Lines           []Line `json:"lines"`
	DiscountPercent int    `json:"discount_percent"`
}

// Subtotal sums the lines before the discount.
func (o Order) Subtotal() (int64, error) {
	if len(o.Lines) == 0 {
		return 0, ErrEmpty
	}
	var sum int64
	for _, line := range o.Lines {
		if line.Quantity <= 0 {
			return 0, fmt.Errorf("line %s: quantity must be positive", line.SKU)
		}
		sum += int64(line.Quantity) * line.UnitPrice
	}
	return sum, nil
}

// Total applies the discount to the subtotal.
func (o Order) Total() (int64, error) {
	subtotal, err := o.Subtotal()
	if err != nil {
		return 0, err
	}
	return ApplyDiscount(subtotal, o.DiscountPercent), nil
}
