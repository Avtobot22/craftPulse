package main

import (
	"log"
	"net/http"
	"time"

	"demo.invalid/orders/internal/config"
	"demo.invalid/orders/internal/httpapi"
)

func main() {
	cfg := config.Load()
	server := &http.Server{
		Addr:              cfg.Addr,
		Handler:           httpapi.NewHandler(),
		ReadHeaderTimeout: 5 * time.Second,
	}
	log.Fatal(server.ListenAndServe())
}
