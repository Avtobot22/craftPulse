package export

import (
	"encoding/csv"
	"io"
	"strconv"
)

// Row is one payroll line; the amount is in kopecks.
type Row struct {
	EmployeeID string
	Amount     int64
}

// WriteCSV writes the rows with a header line.
func WriteCSV(w io.Writer, rows []Row) error {
	out := csv.NewWriter(w)
	if err := out.Write([]string{"employee_id", "amount"}); err != nil {
		return err
	}
	for _, row := range rows {
		if err := out.Write([]string{row.EmployeeID, strconv.FormatInt(row.Amount, 10)}); err != nil {
			return err
		}
	}
	out.Flush()
	return out.Error()
}
